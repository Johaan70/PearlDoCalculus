def Identification.Identifiable.{u_1, u_2} : {V : Type u_1} →
  [inst : DecidableEq V] →
    [inst_1 : Fintype V] → PearlDoCalculus.DAG V → (V → Type u_2) → Finset V → Finset V → Finset V → Prop :=
fun {V} [DecidableEq V] [Fintype V] G α O X Y =>
  ∀ (M₁ M₂ : G.CausalModel α),
    M₁.marginal O = M₂.marginal O →
      (∀ (a : PearlDoCalculus.DAG.CausalModel.Assignment O), (M₁.marginal O) a ≠ 0) →
        ∀ (t : PearlDoCalculus.DAG.CausalModel.Assignment (Y ∪ X)),
          ((DoCalculus.doModel M₁ X (PearlDoCalculus.DAG.CausalModel.Assignment.restrict ⋯ t)).marginal Y)
              (PearlDoCalculus.DAG.CausalModel.Assignment.restrict ⋯ t) =
            ((DoCalculus.doModel M₂ X (PearlDoCalculus.DAG.CausalModel.Assignment.restrict ⋯ t)).marginal Y)
              (PearlDoCalculus.DAG.CausalModel.Assignment.restrict ⋯ t)
def Identification.CFactor.fill.{u_1, u_2} : {V : Type u_1} →
  [inst : DecidableEq V] →
    [inst_1 : Fintype V] →
      {α : V → Type u_2} →
        [∀ (v : V), Nonempty (α v)] →
          (U T : Finset V) →
            PearlDoCalculus.DAG.CausalModel.Assignment Uᶜ →
              PearlDoCalculus.DAG.CausalModel.Assignment (U ∩ T) →
                PearlDoCalculus.DAG.CausalModel.Assignment Finset.univ :=
fun {V} [DecidableEq V] [Fintype V] {α} [∀ (v : V), Nonempty (α v)] U T a w v =>
  if h0 : ↑v ∈ U then if h1 : ↑v ∈ T then w ⟨↑v, ⋯⟩ else Classical.arbitrary (α ↑v) else a ⟨↑v, ⋯⟩
def Identification.CFactor.Q.{u_1, u_2} : {V : Type u_1} →
  [inst : DecidableEq V] →
    [inst_1 : Fintype V] →
      {G : PearlDoCalculus.DAG V} →
        {α : V → Type u_2} →
          [∀ (v : V), Nonempty (α v)] →
            G.CausalModel α → (U : Finset V) → Finset V → PearlDoCalculus.DAG.CausalModel.Assignment Uᶜ → ENNReal :=
fun {V} [DecidableEq V] [Fintype V] {G} {α} [∀ (v : V), Nonempty (α v)] M U T a =>
  ∑' (w : PearlDoCalculus.DAG.CausalModel.Assignment (U ∩ T)),
    ∏ v ∈ T, PearlDoCalculus.kfac M Finset.univ (Identification.CFactor.fill U T a w) v
def Identification.CFactor.IsClosedBlock.{u_1} : {V : Type u_1} →
  [inst : DecidableEq V] → [inst_1 : Fintype V] → PearlDoCalculus.DAG V → Finset V → Finset V → Prop :=
fun {V} [DecidableEq V] [Fintype V] G U S => ∀ u ∈ U, ∀ (v : V), G.edge u v → (u ∈ S ↔ v ∈ S)
def Identification.CFactor.latStep.{u_1} : {V : Type u_1} →
  [inst : DecidableEq V] → [inst_1 : Fintype V] → PearlDoCalculus.DAG V → Finset V → V → V → Prop :=
fun {V} [DecidableEq V] [Fintype V] G U a b => a ∈ U ∧ G.edge a b ∨ b ∈ U ∧ G.edge b a
